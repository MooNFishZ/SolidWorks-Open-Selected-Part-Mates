Option Explicit

Dim swApp As SldWorks.SldWorks
Dim swModel As ModelDoc2
Dim swSelMgr As SelectionMgr
Dim swComp As Component2

Sub main()
    On Error Resume Next

    Set swApp = Application.SldWorks
    Set swModel = swApp.ActiveDoc

    If swModel Is Nothing Then Exit Sub

    Set swSelMgr = swModel.SelectionManager
    If swSelMgr Is Nothing Then Exit Sub
    If swSelMgr.GetSelectedObjectCount2(-1) = 0 Then Exit Sub

    Set swComp = swSelMgr.GetSelectedObjectsComponent4(1, -1)
    If swComp Is Nothing Then Exit Sub

    Dim targetComp As Component2
    Set targetComp = ClimbToTopLevel(swComp)

    MatesForm.LoadMates targetComp, swModel
    MatesForm.Show vbModeless

    On Error GoTo 0
End Sub

' Podnimayemsya po derevu do komponenta verkhnego urovnya tekushchey sborki
' <-- ïîäíèìàåìñÿ ïî äåðåâó äî êîìïîíåíòà âåðõíåãî óðîâíÿ òåêóùåé ñáîðêè
Private Function ClimbToTopLevel(comp As Component2) As Component2
    On Error GoTo Fail

    Dim c As Component2
    Set c = comp
    Do While Not c.GetParent Is Nothing
        Set c = c.GetParent
    Loop
    Set ClimbToTopLevel = c
    Exit Function

Fail:
    Set ClimbToTopLevel = comp
End Function
